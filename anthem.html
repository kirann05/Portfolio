Anomaly Detection Pipeline 

class TransactionAnomalyDetector:
    def __init__(self, model_path="models/anomaly_xgb_v3.pkl"):
        self.model = joblib.load(model_path)
        self.scaler = StandardScaler()
        self.feature_cols = [
            "amount", "merchant_risk_score", "txn_velocity_1h",
            "geo_deviation", "time_since_last_txn", "avg_amount_30d"
        ]

    def detect(self, transaction_batch: pd.DataFrame):
        """Score transactions for anomaly likelihood"""
        features = transaction_batch[self.feature_cols]
        scaled = self.scaler.transform(features)
        scores = self.model.predict_proba(scaled)[:, 1]

        transaction_batch["anomaly_score"] = scores
        flagged = transaction_batch[scores > 0.85]

        return {
            "total_processed": len(transaction_batch),
            "flagged_count": len(flagged),
            "flagged_transactions": flagged.to_dict("records"),
            "avg_confidence": float(scores[scores > 0.85].mean())
        }

Optimized Financial Query (40% Faster) 

-- Optimized credit risk aggregation query
-- Snowflake warehouse, 40% performance gain
WITH daily_risk_metrics AS (
    SELECT
        entity_id,
        rating_date,
        credit_score,
        AVG(credit_score) OVER (
            PARTITION BY entity_id
            ORDER BY rating_date
            ROWS BETWEEN 90 PRECEDING AND CURRENT ROW
        ) AS rolling_avg_90d,
        STDDEV(credit_score) OVER (
            PARTITION BY entity_id
            ORDER BY rating_date
            ROWS BETWEEN 30 PRECEDING AND CURRENT ROW
        ) AS volatility_30d
    FROM credit_ratings
    WHERE rating_date >= DATEADD('year', -2, CURRENT_DATE)
)
SELECT entity_id, rating_date, credit_score,
    rolling_avg_90d, volatility_30d,
    CASE
        WHEN credit_score < rolling_avg_90d - (2 * volatility_30d)
        THEN 'ELEVATED_RISK'
        ELSE 'NORMAL'
    END AS risk_flag
FROM daily_risk_metrics;
